Complete Official No-Demo Sources

The startup file keeps the game story visible. The shared model file teaches Enum, Type, Class, New, Me, methods, properties, and private state.

Program-NoDemo.smileSnakeModel.smileExact repository sources

Why two files?

Program-NoDemo.smile owns score, food policy, high score, input, timing, audio, screens, and the main loop. SnakeModel.smile owns only the snake’s segments, length, direction, turns, movement, growth, and collision queries. This is focused object-oriented design—not “everything is an object.”

Program-NoDemo.smile

games\Snake\Program-NoDemo.smile — complete source
' SMILE 2.0 Snake - student edition without attract/demo code.

Option Explicit

Const CanvasWidth = 960
Const CanvasHeight = 540
Const FieldX = 20
Const FieldY = 30
Const FieldWidth = 700
Const FieldHeight = 480
Const GridWidth = 35
Const GridHeight = 24
Const CellSize = 20
Const StartDelay = 100
Const MinimumDelay = 45

Dim Player As New Snake()
Dim Food As GridPoint
Dim State As GameState
Dim HighScore As Number
Dim Score As Number
Dim MoveDelay As Number
Dim NextMoveTime As Number
Dim Key As Number

With Food
    .X = 0
    .Y = 0
End With

State = GameState.Title
Score = 0
MoveDelay = StartDelay
NextMoveTime = 0
Key = KEY_NONE

'----------------------------------------------------------------------------------------------------

Load HighScore From "HighScore" Default 0

Game Window "SMILE 2.0 Snake"

Call EnterTitle()

Do
    Get Key Key

    If State = GameState.Title Then

        If Key = KEY_ESCAPE Then
            End Program
        Else If Key = KEY_ENTER Then
            Call ResetGame()
        End If

        Call DrawTitle()

    Else If State = GameState.Playing Then

        If Key = KEY_ESCAPE Then
            End Program
        End If

        If Key = KEY_W Or Key = KEY_UP Then
            Call Player.TryTurn(MoveDirection.Up)
        Else If Key = KEY_S Or Key = KEY_DOWN Then
            Call Player.TryTurn(MoveDirection.Down)
        Else If Key = KEY_A Or Key = KEY_LEFT Then
            Call Player.TryTurn(MoveDirection.Left)
        Else If Key = KEY_D Or Key = KEY_RIGHT Then
            Call Player.TryTurn(MoveDirection.Right)
        End If

        If State = GameState.Playing And Timer() >= NextMoveTime Then
            Call MoveSnake()
        End If

        Call DrawBoard()

        If State = GameState.GameOver Then
            Call DrawGameOver()
        End If

    Else

        If Key = KEY_ESCAPE Then
            End Program
        Else If Key = KEY_ENTER Then
            Call ResetGame()
        End If

        If State = GameState.Title Then
            Call DrawTitle()
        Else

            Call DrawBoard()
            Call DrawGameOver()
        End If

    End If

    Show Screen

Loop Until Game_Closed() = True

Stop Music
Stop Sound

'----------------------------------------------------------------------------------------------------

Sub SpawnFood()

    Dim FoodValid As Boolean
    Dim NewX As Number
    Dim NewY As Number

    FoodValid = False

    Do
        Random NewX From 0 To GridWidth - 1
        Random NewY From 0 To GridHeight - 1

        Food.X = NewX
        Food.Y = NewY

        FoodValid = Not Player.Contains(Food.X, Food.Y)

    Loop Until FoodValid = True

End Sub

Sub ResetPlay()

    Call Player.Reset()

    Score = 0
    MoveDelay = StartDelay

    Call SpawnFood()

    NextMoveTime = Timer() + StartDelay
    State = GameState.Playing

    Play Sound "Assets\Start.wav"

End Sub

Sub EndRound()

    Stop Music
    State = GameState.GameOver

    If Score > HighScore Then
        HighScore = Score
        Save HighScore To "HighScore"
    End If

    Play Sound "Assets\GameOver.wav"

End Sub

Sub MoveSnake()

    Dim Collision As Boolean
    Dim Grew As Boolean

    Call Player.Move()

    Collision = Player.HitWall(GridWidth, GridHeight)
    Collision = Collision Or Player.HitSelf()

    If Collision Then

        Call EndRound()

    Else If Player.HeadX = Food.X And Player.HeadY = Food.Y Then

        Grew = Player.Grow()

        If Grew Then
            Score = Score + 10
            MoveDelay = Max(MinimumDelay, StartDelay - (Score / 50) * 4)
        End If

        Play Sound "Assets\Eat.wav"
        Call SpawnFood()

    End If

    NextMoveTime = Timer() + MoveDelay

End Sub

Sub DrawBoard()

    Dim I As Number
    Dim SegmentX As Number
    Dim SegmentY As Number

    Clear Rgb(10, 18, 35)
    Fill Rectangle FieldX, FieldY, FieldWidth, FieldHeight, BLACK
    Draw Rectangle FieldX, FieldY, FieldWidth, FieldHeight, Rgb(40, 70, 95)

    For I = 0 To Player.Length - 1

        SegmentX = Player.SegmentX(I)
        SegmentY = Player.SegmentY(I)

        If (SegmentX >= 0 And
            SegmentX < GridWidth And
            SegmentY >= 0 And
            SegmentY < GridHeight) Then
            Fill Rectangle FieldX + SegmentX * CellSize + 2, FieldY + SegmentY * CellSize + 2, 16, 16, Rgb(60, 230, 75)
            Draw Rectangle FieldX + SegmentX * CellSize + 2, FieldY + SegmentY * CellSize + 2, 16, 16, Rgb(20, 105, 35)
        End If

    End For

    Fill Rectangle FieldX + Food.X * CellSize + 3, FieldY + Food.Y * CellSize + 3, 14, 14, Rgb(245, 45, 55)
    Draw Rectangle FieldX + Food.X * CellSize + 3, FieldY + Food.Y * CellSize + 3, 14, 14, Rgb(120, 15, 25)

    Draw Text "SMILE Snake" At 835, 55 Size 27 Color LIGHT_GREEN Centered
    Draw Text "Score" At 755, 115 Size 20 Color LIGHT_GRAY
    Draw Number Score At 755, 142 Size 34 Color YELLOW
    Draw Text "High Score" At 755, 205 Size 20 Color LIGHT_GRAY
    Draw Number HighScore At 755, 232 Size 34 Color CYAN
    Draw Text "WASD / Arrows" At 835, 330 Size 16 Color WHITE Centered
    Draw Text "Move The Snake" At 835, 355 Size 16 Color LIGHT_GRAY Centered
    Draw Text "Alt+Enter" At 835, 415 Size 16 Color WHITE Centered
    Draw Text "Full Screen" At 835, 440 Size 16 Color LIGHT_GRAY Centered
    Draw Text "Escape Exits" At 835, 490 Size 16 Color LIGHT_GRAY Centered

End Sub

SnakeModel.smile

games\Snake\SnakeModel.smile — complete source
' Shared value and reference models for both Snake startup variants.

Option Explicit

Enum GameState
    Title
    Playing
    GameOver
End Enum

Enum MoveDirection
    Up
    Down
    Left
    Right
End Enum

Type GridPoint
    X As Number
    Y As Number
End Type

Class Snake

    Private SegmentXs[840] As Number
    Private SegmentYs[840] As Number
    Private CurrentLength As Number
    Private CurrentDirection As MoveDirection
    Private RequestedDirection As MoveDirection
    Private TailX As Number
    Private TailY As Number

    Public Sub New()

        Call Me.Reset()

    End Sub

    Public Sub Reset()

        Me.CurrentLength = 5
        Me.CurrentDirection = MoveDirection.Right
        Me.RequestedDirection = MoveDirection.Right
        Me.TailX = 0
        Me.TailY = 0

        Me.SegmentXs[0] = 17
        Me.SegmentYs[0] = 12
        Me.SegmentXs[1] = 16
        Me.SegmentYs[1] = 12
        Me.SegmentXs[2] = 15
        Me.SegmentYs[2] = 12
        Me.SegmentXs[3] = 14
        Me.SegmentYs[3] = 12
        Me.SegmentXs[4] = 13
        Me.SegmentYs[4] = 12

    End Sub

    Public Sub TryTurn(Direction As MoveDirection)

        If Not Me.IsReverse(Direction) Then
            Me.RequestedDirection = Direction
        End If

    End Sub

    Public Function CanTurn(Direction As MoveDirection) As Boolean

        Dim Result As Boolean

        Result = Not Me.IsReverse(Direction)

        Return Result

    End Function

    Public Sub Move()

        Dim I As Number

        Me.CurrentDirection = Me.RequestedDirection
        Me.TailX = Me.SegmentXs[Me.CurrentLength - 1]
        Me.TailY = Me.SegmentYs[Me.CurrentLength - 1]

        For I = Me.CurrentLength - 1 Down To 1
            Me.SegmentXs[I] = Me.SegmentXs[I - 1]
            Me.SegmentYs[I] = Me.SegmentYs[I - 1]
        End For

        Select Case Me.CurrentDirection
            Case MoveDirection.Up
                Me.SegmentYs[0] = Me.SegmentYs[0] - 1

            Case MoveDirection.Down
                Me.SegmentYs[0] = Me.SegmentYs[0] + 1

            Case MoveDirection.Left
                Me.SegmentXs[0] = Me.SegmentXs[0] - 1

            Case MoveDirection.Right
                Me.SegmentXs[0] = Me.SegmentXs[0] + 1
        End Select

    End Sub

    Public Function Grow() As Boolean

        If Me.CurrentLength >= 840 Then
            Return False
        End If

        Me.SegmentXs[Me.CurrentLength] = Me.TailX
        Me.SegmentYs[Me.CurrentLength] = Me.TailY
        Me.CurrentLength = Me.CurrentLength + 1

        Return True

    End Function

    Public Function Contains(GridX As Number, GridY As Number) As Boolean

        Dim Result As Boolean
        Dim I As Number

        Result = False

        For I = 0 To Me.CurrentLength - 1

            If Me.SegmentXs[I] = GridX And Me.SegmentYs[I] = GridY Then
                Result = True
                Exit For
            End If

        End For

        Return Result

    End Function

    Public Function CellBlocked(
        GridX As Number,
        GridY As Number,
        GridWidth As Number,
        GridHeight As Number,
        AllowTail As Boolean
    ) As Boolean

        Dim Result As Boolean
        Dim I As Number

        Result = False

        If (GridX < 0 Or
            GridX >= GridWidth Or
            GridY < 0 Or
            GridY >= GridHeight) Then
            Return True
        End If

        For I = 0 To Me.CurrentLength - 1

            If Me.SegmentXs[I] = GridX And Me.SegmentYs[I] = GridY Then

                If Not AllowTail Or I <> Me.CurrentLength - 1 Then
                    Result = True
                End If

                Exit For

            End If

        End For

        Return Result

    End Function

    Public Function HitSelf() As Boolean

        Dim Result As Boolean
        Dim I As Number

        Result = False

        For I = 1 To Me.CurrentLength - 1

            If Me.SegmentXs[0] = Me.SegmentXs[I] And Me.SegmentYs[0] = Me.SegmentYs[I] Then
                Result = True
                Exit For
            End If

        End For

        Return Result

    End Function

    Public Function HitWall(GridWidth As Number, GridHeight As Number) As Boolean

        Dim Result As Boolean

        Result = (Me.SegmentXs[0] < 0 Or
            Me.SegmentXs[0] >= GridWidth Or
            Me.SegmentYs[0] < 0 Or
            Me.SegmentYs[0] >= GridHeight)

        Return Result

    End Function

    Public Function SegmentX(Index As Number) As Number

        Dim Result As Number

        Result = 0

        If Index >= 0 And Index < Me.CurrentLength Then
            Result = Me.SegmentXs[Index]
        End If

        Return Result

    End Function

    Public Function SegmentY(Index As Number) As Number

        Dim Result As Number

        Result = 0

        If Index >= 0 And Index < Me.CurrentLength Then
            Result = Me.SegmentYs[Index]
        End If

        Return Result

    End Function

    Public Property Length As Number

        Get

            Dim Result As Number

            Result = Me.CurrentLength

            Return Result

        End Get

    End Property

    Public Property HeadX As Number

        Get

            Dim Result As Number

            Result = Me.SegmentXs[0]

            Return Result

        End Get

    End Property

    Public Property HeadY As Number

        Get

            Dim Result As Number

            Result = Me.SegmentYs[0]

            Return Result

        End Get

    End Property

    Public Property Direction As MoveDirection

        Get

            Dim Result As MoveDirection

            Result = Me.CurrentDirection

            Return Result

        End Get

    End Property

    Private Function IsReverse(Direction As MoveDirection) As Boolean

        Dim Result As Boolean

        Result = ((Direction = MoveDirection.Up And Me.CurrentDirection = MoveDirection.Down) Or
            (Direction = MoveDirection.Down And Me.CurrentDirection = MoveDirection.Up) Or
            (Direction = MoveDirection.Left And Me.CurrentDirection = MoveDirection.Right) Or
            (Direction = MoveDirection.Right And Me.CurrentDirection = MoveDirection.Left))

        Return Result

    End Function

End Class
